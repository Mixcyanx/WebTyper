import java.util.*;
public class main {
public static void main(String[] args) {
Scanner in=new Scanner(System.in);
while(true) {
String s=in.next();
if(s.equals("0")) break;
int degree=0;
int sum=0;
for(int i=0;i<s.length();i++) {
sum+=s.charAt(i)-'0';}
degree++;
while(sum>=10) {
int temp=sum;
sum=0;
while(temp>0) {
sum+=temp%10;
temp/=10;}
degree++;}
if(sum%9==0) {
System.out.print(s+" is a multiple of 9 and has 9-degree "+degree+".\r\n");
} else {
System.out.print(s+" is not a multiple of 9.\r\n");}}}}
