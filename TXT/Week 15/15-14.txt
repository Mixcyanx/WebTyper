import java.util.*;
import java.io.*;
public class main {
public static void main(String[] args) throws Exception {
FastScanner in=new FastScanner(System.in);
StringBuilder out=new StringBuilder();
int T=in.nextInt();
for(int t=0;t<T;t++) {
int M=in.nextInt();
int[] a=new int[M];
for(int i=0;i<M;i++) a[i]=in.nextInt()+in.nextInt();
int carry=0;
for(int i=M-1;i>=0;i--) {
int sum=a[i]+carry;
a[i]=sum%10;
carry=sum/10;
}
if(carry>0) out.append(carry);
for(int i=0;i<M;i++) out.append(a[i]);
out.append("\r\n");
if(t<T-1) out.append("\r\n");
}
System.out.print(out.toString());
}
static class FastScanner {
private final InputStream in;
private final byte[] buffer=new byte[1<<16];
private int ptr=0,len=0;
FastScanner(InputStream is) {
in=is;
}
private int read() throws IOException {
if(ptr>=len) {
len=in.read(buffer);
ptr=0;
if(len<=0) return -1;
}
return buffer[ptr++];
}
int nextInt() throws IOException {
int c=read();
while(c<=32&&c!=-1) c=read();
int val=0;
while(c>32&&c!=-1) {
val=val*10+c-'0';
c=read();
}
return val;
}
}
}
