import java.util.*;

public class main {
public static void main(String[] args) {
Scanner in=new Scanner(System.in);

int n=in.nextInt();
int[] a=new int[n];
boolean[] vis=new boolean[n];

for(int i=0;i<n;i++) {
a[i]=in.nextInt();
}

int count=0;

for(int i=0;i<n;i++) {

if(!vis[i]) {

int now=i;

while(!vis[now]) {
vis[now]=true;
now=a[now];
}

count++;
}
}

System.out.println(count);
}
}
