import java.util.*;

public class main {
public static void main(String[] args) {
Scanner in=new Scanner(System.in);

int a=in.nextInt();
int b=in.nextInt();
int c=in.nextInt();

boolean ok=false;

if((a!=0&&b!=0?1:0)==c) {
System.out.println("AND");
ok=true;
}

if((a!=0||b!=0?1:0)==c) {
System.out.println("OR");
ok=true;
}

if(((a!=0)^(b!=0)?1:0)==c) {
System.out.println("XOR");
ok=true;
}

if(!ok) {
System.out.println("IMPOSSIBLE");
}
}
}
