import java.util.*;

interface CanvasIn {
boolean[][] getCanvas();
void setCanvas(boolean[][] canvas);
public abstract void showCanvas();
}

class MyCanvas implements CanvasIn {
boolean[][] canvas = new boolean[25][40];
Scanner in = new Scanner(System.in);

public boolean[][] getCanvas() {
return canvas;}

public void setCanvas(boolean[][] canvas) {
this.canvas = canvas;}

public void readDot() {
for(int i=0;i<25;i++) {
for(int j=0;j<40;j++) {
int n=in.nextInt();
canvas[i][j]=(n==1);
}}}

public void drawDot(int x,int y) {
canvas[x][y]=true;}

public void eraseDot(int x,int y) {
canvas[x][y]=false;}

public void showCanvas() {
for(int i=0;i<25;i++) {
for(int j=0;j<40;j++) {
System.out.print(canvas[i][j]?"A":"0");
if(j<40) System.out.print(" ");
}
System.out.println();
}}}

class main  {
public static void main (String[] args) { 
MyCanvas c = new MyCanvas(); 
c.readDot();
for( int k = 0 ; k < 20 ; k++ )  { 
c.drawDot(k, k);
}
c.showCanvas();
}}
