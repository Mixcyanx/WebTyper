import java.util.*;
class Animal{
public void run(int a){
System.out.println("Animal's running speed is "+a);
}}
class Dog extends Animal{
public void run(int a){
System.out.println("Dog's running speed is "+a);
}
public void run(){
System.out.println("Dog's default running speed is 20");
}}
public class main {
public static void main(String[] args) {
Scanner in = new Scanner(System.in);
int a = in.nextInt();
int b = in.nextInt();
Animal animal = new Animal();
animal.run(a);
Dog dog = new Dog();
dog.run(b);
dog.run();
}}
