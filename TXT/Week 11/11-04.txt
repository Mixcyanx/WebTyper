import java.util.*;
public class main {
public static void main(String[] args) {
Scanner in=new Scanner(System.in);
int a,b;
double c,d;
a=in.nextInt();
b=in.nextInt();
c=in.nextDouble();
d=in.nextDouble();
MyTestClass myObject=new MyTestClass();
System.out.println(MyTestClass.add(a,b));
System.out.println(MyTestClass.add(c,d));
System.out.println(myObject.add(a,b,c));
System.out.println(myObject.add(a,b,c,d));
}
}
class MyTestClass {
public static int add(int a,int b) {
return a+b;
}
public static double add(double a,double b) {
return a+b;
}
public double add(int a,int b,double c) {
return a+b+c;
}
public double add(int a,int b,double c,double d) {
return a+b+c+d;
}
}
