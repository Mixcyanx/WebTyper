protected String phone;

return name;
}
}

class MyDate {
private int year;
private int month;
private int day;
public MyDate() {
year=2024;
month=1;
day=1;
}
public MyDate(int year,int month,int day) {
this.year=year;
this.month=month;
this.day=day;
}
public int getYear() {
return year;
}
public int getMonth() {
return month;
}
public int getDay() {
return day;
}
