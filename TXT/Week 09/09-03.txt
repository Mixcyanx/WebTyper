import java.util.*;
public class main {
public static void main(String[] args) {
Scanner in=new Scanner(System.in);
int n=in.nextInt();
for(int i=0;i<n;i++){
String a=in.next();
String b=in.next();
String p="";
int m=Math.min(a.length(),b.length());
for(int j=0;j<m;j++){
if(a.charAt(j)==b.charAt(j))p+=a.charAt(j);
else break;}
if(p.length()>0)System.out.print("The common prefix is "+p+"\r\n");
else System.out.print("No common prefix\r\n");}}}
