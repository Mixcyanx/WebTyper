import java.util.*;
public class main {
public static void main(String[] args) {
Scanner in=new Scanner(System.in);
if(in.hasNextLine()){
String s=in.nextLine().toUpperCase();
int[] c=new int[26];
count(s,0,c);
print(c,0); }}
public static void count(String s,int i,int[] c){
if(i<s.length()){
char ch=s.charAt(i);
if(ch>='A'&&ch<='Z'){
c[ch-'A']++; }
count(s,i+1,c); }}
public static void print(int[] c,int i){
if(i<26){
if(c[i]>0){
System.out.print((char)(i+'A')+": "+c[i]+"\r\n"); }
print(c,i+1); }}}
